```typescript
// packages/engine/tests/hooks/stop-hook-v24.test.ts
import { describe, it, expect, beforeEach, afterEach } from 'vitest'
import { execSync } from 'child_process'
import { writeFileSync, mkdirSync, rmSync, mkdtempSync, utimesSync, existsSync } from 'fs'
import { resolve, join } from 'path'
import { tmpdir } from 'os'

const HOOK = resolve(__dirname, '../../hooks/stop-dev.sh')

/** v24 接口：用 env var 传 session_id，不 pipe stdin */
function runHookV24(testRepo: string, sessionId: string, extraEnv: Record<string, string> = {}): {
  stdout: string; stderr: string; code: number
} {
  const envStr = Object.entries({ CLAUDE_HOOK_CWD: testRepo, CLAUDE_HOOK_SESSION_ID: sessionId, ...extraEnv })
    .map(([k, v]) => `${k}='${v}'`).join(' ')
  try {
    const out = execSync(`cd ${testRepo} && ${envStr} bash ${HOOK}`, { encoding: 'utf8' })
    return { stdout: out, stderr: '', code: 0 }
  } catch (e: any) {
    return { stdout: e.stdout || '', stderr: e.stderr || '', code: e.status ?? 1 }
  }
}

function makeLight(lightsDir: string, sidShort: string, branch: string, guardianPid = 99999, ageSeconds = 0) {
  mkdirSync(lightsDir, { recursive: true })
  const f = join(lightsDir, `${sidShort}-${branch}.live`)
  writeFileSync(f, JSON.stringify({
    session_id: `${sidShort}-full-uuid`,
    branch,
    worktree_path: `/tmp/wt-${branch}`,
    started_at: new Date().toISOString(),
    host: 'test-host',
    guardian_pid: guardianPid,
  }))
  if (ageSeconds > 0) {
    const t = (Date.now() - ageSeconds * 1000) / 1000
    utimesSync(f, t, t)
  }
  return f
}

/** 在 testRepo 中放置 fake devloop-check.sh，使 classify_session 返回指定 JSON */
function mockClassifySession(testRepo: string, resultJson: string) {
  const libDir = join(testRepo, 'packages/engine/lib')
  mkdirSync(libDir, { recursive: true })
  writeFileSync(join(libDir, 'devloop-check.sh'), `#!/usr/bin/env bash
classify_session() {
  echo '${resultJson.replace(/'/g, "'\\''")}'
  return 0
}
log_hook_decision() { :; }
`, { mode: 0o755 })
}

describe('stop-dev.sh v24 新场景（T13-T16）', () => {
  let testRepo: string
  let lightsDir: string

  beforeEach(() => {
    testRepo = mkdtempSync(join(tmpdir(), 'hookv24-'))
    execSync(`cd ${testRepo} && git init -q && git -c user.email=t@t -c user.name=t commit --allow-empty -m init -q`)
    lightsDir = join(testRepo, '.cecelia/lights')
  })

  afterEach(() => {
    rmSync(testRepo, { recursive: true, force: true })
  })

  it('T13: 灯亮 + classify=blocked + action 有值 → block reason 包含 action 文本', () => {
    makeLight(lightsDir, 'abc12345', 'cp-test')
    mockClassifySession(testRepo, JSON.stringify({
      status: 'blocked',
      reason: 'CI 失败（failure）',
      action: 'gh run view 12345 --log-failed',
    }))
    const r = runHookV24(testRepo, 'abc12345-full-uuid')
    expect(r.stdout).toMatch(/"decision"\s*:\s*"block"/)
    expect(r.stdout).toMatch(/gh run view 12345/)
    expect(r.stdout).toMatch(/CI 失败/)
  })

  it('T14: 灯亮 + classify=done → guardian 被 kill，灯文件被删，exit 0', () => {
    // 启动真实子进程作为 guardian
    const guardianProc = execSync('sleep 999 & echo $!', { encoding: 'utf8' }).trim()
    const guardianPid = parseInt(guardianProc)
    const lightFile = makeLight(lightsDir, 'abc12345', 'cp-test', guardianPid)
    mockClassifySession(testRepo, JSON.stringify({ status: 'done', reason: 'cleanup_done' }))

    const r = runHookV24(testRepo, 'abc12345-full-uuid')

    expect(r.code).toBe(0)
    expect(r.stdout).not.toMatch(/decision.*block/)
    // 灯文件已被删
    expect(existsSync(lightFile)).toBe(false)
    // guardian 进程已被 kill（给 1s 传播时间）
    execSync(`sleep 0.5`)
    expect(() => execSync(`kill -0 ${guardianPid} 2>/dev/null`)).toThrow()
  })

  it('T15: 灯亮 + classify=not-dev → exit 0 放行（孤儿灯不影响决策）', () => {
    makeLight(lightsDir, 'abc12345', 'cp-test')
    mockClassifySession(testRepo, JSON.stringify({ status: 'not-dev', reason: '主分支放行' }))
    const r = runHookV24(testRepo, 'abc12345-full-uuid')
    expect(r.code).toBe(0)
    expect(r.stdout).not.toMatch(/decision.*block/)
  })

  it('T16: session_id 空（无 env var）→ exit 0 放行', () => {
    makeLight(lightsDir, 'abc12345', 'cp-test')
    const r = runHookV24(testRepo, '')  // 空 session_id
    expect(r.code).toBe(0)
    expect(r.stdout).not.toMatch(/decision.*block/)
  })
})
```
